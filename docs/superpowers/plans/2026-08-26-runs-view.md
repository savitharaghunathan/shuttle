# Runs View + Metadata Implementation Plan (Plan 1 of 2)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Persist run metadata (agent, model, issue_ref, parent_run_id), fix `runOpts` loss across sessions, and add `shuttle runs list` (+ `--json`) and `GET /v1/runs` with filters and run-ID prefix resolution.

**Architecture:** Additive SQLite migration on `runs`; `Store.ListRuns(opts)` and `Store.ResolveRunID(prefix)`; persist the *resolved* worker in `StartRun` and reload it in `EnsurePipeline`; new `internal/cli/runs.go` for the `runs list` command; extend the API list/get/create JSON. Current-stage and duration are derived at query time using template order.

**Tech Stack:** Go, `modernc.org/sqlite`, cobra, chi, `text/tabwriter`.

**Spec:** [`docs/superpowers/specs/2026-08-26-run-recovery-design.md`](../specs/2026-08-26-run-recovery-design.md) (rev 3). This plan implements the **Plan 1** scope only; recovery (`resume`/`retry`) is Plan 2.

## Global Constraints

- Go module: `github.com/savitharaghunathan/shuttle`
- Single SQLite writer per DB file
- Prefer MIT / Apache-2.0 OSS libraries
- Git commits: `git commit -s` plus body line `Assisted-by: Cursor <cursoragent@cursor.com>`. Do **not** add `Co-authored-by: Cursor`. Ask user for attribution before each commit.
- Stage ordering: never infer pipeline order from DB row order (`GetStageRuns` is `ORDER BY id`). Always iterate `pl.Stages` and join status via a `stage_id → StageRun` map.
- Test doubles live in `internal/worker/workertest` (test-only package, excluded from the production binary). Use `workertest.NewFakeWorker(map[string]worker.Result)` (keyed by **stage id**). The `newTestEngine` helper already registers it as `"fake"` and sets it as `DefaultWorker`; stages with no explicit `Worker` resolve to it.

```go
var (
	ErrRunIDTooShort  = errors.New("run ID prefix too short (min 8 chars)")
	ErrRunIDNotFound  = errors.New("run not found")
	ErrRunIDAmbiguous = errors.New("ambiguous run ID prefix")
)

func (s *Store) ResolveRunID(prefix string) (string, error) {
	p := strings.ToLower(strings.TrimSpace(prefix))
	if p == "" {
		return "", ErrRunIDNotFound
	}
	// exact match first
	var exact string
	err := s.db.QueryRow(`SELECT id FROM runs WHERE id = ?`, p).Scan(&exact)
	if err == nil {
		return exact, nil
	}
	if !errors.Is(err, sql.ErrNoRows) {
		return "", fmt.Errorf("resolve run id: %w", err)
	}
	if len(p) < 8 {
		return "", ErrRunIDTooShort
	}
	rows, err := s.db.Query(`SELECT id FROM runs WHERE id LIKE ? || '%' ORDER BY id LIMIT 2`, p)
	if err != nil {
		return "", fmt.Errorf("resolve run id: %w", err)
	}
	defer rows.Close()
	var matches []string
	for rows.Next() {
		var id string
		if err := rows.Scan(&id); err != nil {
			return "", err
		}
		matches = append(matches, id)
	}
	if err := rows.Err(); err != nil {
		return "", err
	}
	switch len(matches) {
	case 0:
		return "", ErrRunIDNotFound
	case 1:
		return matches[0], nil
	default:
		return "", fmt.Errorf("%w: %q matches multiple runs", ErrRunIDAmbiguous, p)
	}
}
```

```go
func (s *Store) ListRuns(opts ListRunsOpts) ([]Run, error) {
	limit := opts.Limit
	if limit <= 0 {
		limit = 20
	}
	var where []string
	var args []any
	if len(opts.Statuses) > 0 {
		ph := make([]string, len(opts.Statuses))
		for i, st := range opts.Statuses {
			ph[i] = "?"
			args = append(args, string(st))
		}
		where = append(where, "status IN ("+strings.Join(ph, ",")+")")
	}
	if opts.Agent != "" {
		where = append(where, "agent = ?")
		args = append(args, opts.Agent)
	}
	if opts.ModelUnset {
		where = append(where, "model = ''")
	} else if opts.Model != "" {
		where = append(where, "model = ?")
		args = append(args, opts.Model)
	}
	if opts.Template != "" {
		where = append(where, "template_id = ?")
		args = append(args, opts.Template)
	}
	if opts.IssueRef != "" {
		where = append(where, "issue_ref = ?")
		args = append(args, opts.IssueRef)
	}
	if opts.ParentID != "" {
		where = append(where, "parent_run_id = ?")
		args = append(args, opts.ParentID)
	}
	if !opts.Since.IsZero() {
		where = append(where, "created_at >= ?")
		args = append(args, formatTime(opts.Since))
	}
	q := `SELECT id, template_id, status, cwd, run_dir, agent, model, issue_ref, parent_run_id, created_at, updated_at FROM runs`
	if len(where) > 0 {
		q += " WHERE " + strings.Join(where, " AND ")
	}
	q += " ORDER BY created_at DESC, id DESC LIMIT ?"
	args = append(args, limit)

	rows, err := s.db.Query(q, args...)
	if err != nil {
		return nil, fmt.Errorf("list runs: %w", err)
	}
	defer rows.Close()
	var runs []Run
	for rows.Next() {
		run, err := scanRun(rows)
		if err != nil {
			return nil, err
		}
		runs = append(runs, run)
	}
	return runs, rows.Err()
}
```

```go
func CurrentStage(pl *Pipeline, stages []store.StageRun) string {
	if pl == nil || len(pl.Stages) == 0 {
		return ""
	}
	byID := make(map[string]store.StageRun, len(stages))
	for _, sr := range stages {
		byID[sr.StageID] = sr
	}
	// 1: first running
	for _, st := range pl.Stages {
		if byID[st.ID].Status == store.StageStatusRunning {
			return st.ID
		}
	}
	// 2: first waiting_approval
	for _, st := range pl.Stages {
		if byID[st.ID].Status == store.StageStatusWaitingApproval {
			return st.ID
		}
	}
	// 3: last failed/cancelled/rejected (in pipeline order)
	last := ""
	for _, st := range pl.Stages {
		switch byID[st.ID].Status {
		case store.StageStatusFailed, store.StageStatusCancelled, store.StageStatusRejected:
			last = st.ID
		}
	}
	if last != "" {
		return last
	}
	// 4: first skipped after the initial succeeded prefix (cancel-during-blocked)
	for _, st := range pl.Stages {
		s := byID[st.ID].Status
		if s == store.StageStatusSucceeded {
			continue
		}
		if s == store.StageStatusSkipped {
			return st.ID
		}
		break
	}
	// 5: last succeeded
	lastOk := ""
	for _, st := range pl.Stages {
		if byID[st.ID].Status == store.StageStatusSucceeded {
			lastOk = st.ID
		}
	}
	if lastOk != "" {
		return lastOk
	}
	// 6
	return "—"
}
```

```go
func parseStatuses(csv string) []store.RunStatus {
	var out []store.RunStatus
	for _, part := range strings.Split(csv, ",") {
		p := strings.TrimSpace(part)
		if p != "" {
			out = append(out, store.RunStatus(p))
		}
	}
	return out
}

func parseSince(s string) (time.Time, error) {
	s = strings.TrimSpace(s)
	if len(s) > 1 && s[len(s)-1] == 'd' {
		var days int
		if _, err := fmt.Sscanf(s, "%dd", &days); err == nil {
			return time.Now().Add(-time.Duration(days) * 24 * time.Hour), nil
		}
	}
	if d, err := time.ParseDuration(s); err == nil {
		return time.Now().Add(-d), nil
	}
	if t, err := time.Parse("2006-01-02", s); err == nil {
		return t, nil
	}
	return time.Time{}, fmt.Errorf("invalid --since %q (use 7d, 24h, or 2026-08-01)", s)
}

func shortID(id string) string {
	if len(id) >= 8 {
		return id[:8]
	}
	return id
}

func runDuration(run store.Run) time.Duration {
	terminal := run.Status == store.RunStatusSucceeded ||
		run.Status == store.RunStatusFailed ||
		run.Status == store.RunStatusCancelled
	if terminal {
		return run.UpdatedAt.Sub(run.CreatedAt)
	}
	return time.Since(run.CreatedAt)
}

func runCurrentStage(app *appContext, run store.Run, cache map[string]*pipeline.Pipeline) string {
	pl, ok := cache[run.TemplateID]
	if !ok {
		loaded, err := loadPipelineTemplate(run.TemplateID)
		if err != nil {
			cache[run.TemplateID] = nil
		} else {
			pl = loaded
			cache[run.TemplateID] = loaded
		}
	}
	if pl == nil {
		return "unknown"
	}
	stages, err := app.store.GetStageRuns(run.ID)
	if err != nil {
		return "unknown"
	}
	return pipeline.CurrentStage(pl, stages)
}

func writeRunsTable(out io.Writer, app *appContext, runs []store.Run) error {
	tw := tabwriter.NewWriter(out, 0, 0, 2, ' ', 0)
	fmt.Fprintln(tw, "RUN\tPARENT\tSTATUS\tAGENT\tMODEL\tTEMPLATE\tISSUE\tSTARTED\tDURATION\tSTAGE")
	cache := map[string]*pipeline.Pipeline{}
	for _, run := range runs {
		agent := run.Agent
		if agent == "" {
			agent = "(unknown)"
		}
		model := run.Model
		if model == "" {
			model = "(default)"
		}
		parent := ""
		if run.ParentRunID != "" {
			parent = shortID(run.ParentRunID)
		}
		fmt.Fprintf(tw, "%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\t%s\n",
			shortID(run.ID), parent, run.Status, agent, model,
			run.TemplateID, run.IssueRef,
			run.CreatedAt.Local().Format("2006-01-02 15:04"),
			runDuration(run).Truncate(time.Second),
			runCurrentStage(app, run, cache),
		)
	}
	return tw.Flush()
}

func writeRunsJSON(out io.Writer, app *appContext, runs []store.Run) error {
	type runJSON struct {
		ID           string `json:"id"`
		ParentRunID  string `json:"parent_run_id,omitempty"`
		Status       string `json:"status"`
		Agent        string `json:"agent"`
		Model        string `json:"model"`
		TemplateID   string `json:"template_id"`
		IssueRef     string `json:"issue_ref,omitempty"`
		CreatedAt    string `json:"created_at"`
		UpdatedAt    string `json:"updated_at"`
		WallDuration int64  `json:"wall_duration"`
		CurrentStage string `json:"current_stage"`
	}
	cache := map[string]*pipeline.Pipeline{}
	list := make([]runJSON, 0, len(runs))
	for _, run := range runs {
		list = append(list, runJSON{
			ID: run.ID, ParentRunID: run.ParentRunID, Status: string(run.Status),
			Agent: run.Agent, Model: run.Model, TemplateID: run.TemplateID, IssueRef: run.IssueRef,
			CreatedAt: run.CreatedAt.UTC().Format(time.RFC3339), UpdatedAt: run.UpdatedAt.UTC().Format(time.RFC3339),
			WallDuration: int64(runDuration(run).Seconds()),
			CurrentStage: runCurrentStage(app, run, cache),
		})
	}
	enc := json.NewEncoder(out)
	enc.SetIndent("", "  ")
	return enc.Encode(list)
}
```
